15fn fetch(url: &str) -> anyhow::Result<Vec<u8>> {
16    let output = Command::new("curl")
17        .args([
18            "--fail",
19            "--silent",
20            "--show-error",
21            "--location",
22            "--connect-timeout",
23            "15",
24            "--max-time",
25            "120",
26            "--user-agent",
27            "serenade-rangular-mcp-refresh",
28            url,
29        ])
30        .output()
31        .context("run curl")?;
32    if !output.status.success() {
33        bail!("fetch {url}: {}", String::from_utf8_lossy(&output.stderr));
34    }
35    Ok(output.stdout)
36}

43fn resolve_commit(repository: &str, reference: &str) -> anyhow::Result<String> {
44    let api = format!("https://api.github.com/repos/Interchouette-ITC/{repository}");
45    let commit: Value = serde_json::from_slice(&fetch(&format!("{api}/commits/{reference}"))?)?;
46    let revision = commit["sha"].as_str().context("missing source commit")?;
47    ensure!(
48        revision.len() == 40
49            && revision
50                .bytes()
51                .all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b)),
52        "invalid {repository} source commit"
53    );
54    Ok(revision.to_owned())
55}

57fn latest_release(repository: &str) -> anyhow::Result<Source> {
58    let api = format!("https://api.github.com/repos/Interchouette-ITC/{repository}");
59    let release: Value = serde_json::from_slice(&fetch(&format!("{api}/releases/latest"))?)?;
60    let tag = release["tag_name"]
61        .as_str()
62        .context("missing release tag")?;
63    ensure!(
64        tag.starts_with('v')
65            && tag.len() > 1
66            && tag[1..].bytes().all(|b| b.is_ascii_digit() || b == b'.'),
67        "unsupported {repository} release tag {tag}"
68    );
69    Ok(Source {
70        tag: Some(tag.to_owned()),
71        revision: resolve_commit(repository, tag)?,
72    })
73}

88fn include_source(product: &str, path: &str) -> bool {
89    if path.starts_with("docs/api-rust/")
90        || path.starts_with("docs/astro/src/content/docs/source-guides/")
91        || path == "docs/astro/src/content/docs/components/catalog.md"
92    {
93        return false;
94    }
95    let source_path = Path::new(path);
96    let name = source_path.file_name().and_then(|value| value.to_str());
97    let repository_administration = matches!(
98        name,
99        Some(
100            "CODE_OF_CONDUCT.md"
101                | "CONTRIBUTING.md"
102                | "RELEASING.md"
103                | "SUPPLY-CHAIN.md"
104                | "pull_request_template.md"
105        )
106    ) || (name == Some("SECURITY.md")
107        && source_path.parent() == Some(Path::new("docs")))
108        || path.starts_with("docs/brand/");
109    if repository_administration {
110        return false;
111    }
112    let extension = source_path.extension().and_then(|value| value.to_str());
113    let markdown = matches!(extension, Some("md" | "mdx"));
114    let crate_readme = path.starts_with("crates/") && path.ends_with("/README.md");
115    let documentation = match product {
116        "serenade" => {
117            path.starts_with("docs-dev/")
118                || path.starts_with("docs/astro/src/content/docs/")
119                || (path.starts_with("docs/") && !path.starts_with("docs/astro/"))
120        }
121        "rangular" => path.starts_with("docs/"),
122        _ => false,
123    };
124    let example = match product {
125        "serenade" => path.starts_with("examples/"),
126        "rangular" => path.starts_with("demo-leptos/") || path.starts_with("demo-tauri/"),
127        _ => false,
128    };
129    let generated_example = example
130        && source_path.components().any(|component| {
131            matches!(component, Component::Normal(name) if matches!(name.to_str(), Some("dist" | "target" | "node_modules")))
132        });
133    (markdown && (documentation || crate_readme))
134        || (example
135            && !generated_example
136            && !is_example_manifest(path)
137            && matches!(
138                extension,
139                Some("md" | "rs" | "html" | "scss" | "css" | "js" | "json" | "toml")
140            ))
141}

143fn validate_source(product: &str, path: &str, text: &str) -> anyhow::Result<()> {
144    if product == "serenade" && path == "docs/astro/src/content/docs/how-to/agent-mcp.md" {
145        ensure!(
146            text.contains("cargo install --locked serenade-rangular-mcp"),
147            "upstream agent guide lacks the release install command; refresh after the guide is updated"
148        );
149        ensure!(
150            text.contains("http://127.0.0.1:8765/`"),
151            "upstream agent guide lacks the root HTTP endpoint; refresh after the guide is updated"
152        );
153    }
154    Ok(())
155}

157fn check_published_sources(root: &Path, revision: &str, version: &str) -> anyhow::Result<()> {
158    let output = Command::new(std::env::var_os("CARGO").unwrap_or_else(|| "cargo".into()))
159        .args(["metadata", "--locked", "--format-version", "1"])
160        .arg("--manifest-path")
161        .arg(root.join("Cargo.toml"))
162        .output()
163        .context("run cargo metadata")?;
164    ensure!(
165        output.status.success(),
166        "cargo metadata failed: {}",
167        String::from_utf8_lossy(&output.stderr)
168    );
169    let metadata: Value = serde_json::from_slice(&output.stdout)?;
170    let packages = metadata["packages"]
171        .as_array()
172        .context("missing packages")?;
173    for name in ["rangular-parser", "rangular-expr", "rangular-host"] {
174        let package = packages
175            .iter()
176            .find(|package| {
177                package["name"] == name
178                    && package["version"] == version
179                    && package["source"]
180                        .as_str()
181                        .is_some_and(|source| source.starts_with("registry+"))
182            })
183            .with_context(|| format!("missing locked registry package {name} {version}"))?;
184        let manifest = package["manifest_path"]
185            .as_str()
186            .context("missing manifest path")?;
187        let vcs_path = Path::new(manifest)
188            .parent()
189            .context("package directory")?
190            .join(".cargo_vcs_info.json");
191        let vcs: Value = serde_json::from_slice(
192            &fs::read(&vcs_path).with_context(|| format!("read {}", vcs_path.display()))?,
193        )?;
194        ensure!(
195            vcs["git"]["sha1"].as_str() == Some(revision),
196            "{name} {version} was published from a different revision than the selected docs"
197        );
198    }
199    Ok(())
200}

202fn update_parser(root: &Path, version: &str) -> anyhow::Result<()> {
203    let cargo: toml::Value = toml::from_str(&fs::read_to_string(root.join("Cargo.toml"))?)?;
204    if cargo["dependencies"]["rangular-parser"].as_str() == Some(format!("={version}").as_str()) {
205        return Ok(());
206    }
207    let status = Command::new(std::env::var_os("CARGO").unwrap_or_else(|| "cargo".into()))
208        .current_dir(root)
209        .args(["add", "--package", "serenade-rangular-mcp"])
210        .arg(format!("rangular-parser@={version}"))
211        .arg("--manifest-path")
212        .arg(root.join("Cargo.toml"))
213        .status()
214        .context("update rangular-parser dependency")?;
215    ensure!(
216        status.success(),
217        "failed to update rangular-parser to {version}"
218    );
219    Ok(())
220}

222#[derive(Default)]
223struct Inventory {
224    products: BTreeMap<String, Value>,
225    documents: Vec<Value>,
226    contents: BTreeMap<String, Vec<u8>>,
227}
228
229fn refresh_product(
230    product: &str,
231    repository: &str,
232    source: &Source,
233    inventory: &mut Inventory,
234) -> anyhow::Result<()> {
235    let url = format!(
236        "https://api.github.com/repos/Interchouette-ITC/{repository}/tarball/{}",
237        source.revision
238    );
239    let archive = fetch(&url)?;
240    let mut archive = tar::Archive::new(GzDecoder::new(archive.as_slice()));
241    let before = inventory.documents.len();
242    let mut source_version = None;
243    for item in archive.entries().context("read release archive")? {
244        let mut item = item.context("read archive entry")?;
245        if !item.header().entry_type().is_file() {
246            continue;
247        }
248        let archive_path = item.path()?.into_owned();
249        let mut components = archive_path.components();
250        components.next().context("missing archive root")?;
251        let path = components.as_path();
252        let path = path.to_str().context("non-UTF-8 archive path")?;
253        ensure!(safe_path(path), "unsafe source path: {path}");
254        if path == "Cargo.toml" {
255            let mut manifest = String::new();
256            item.read_to_string(&mut manifest)?;
257            let cargo: toml::Value = toml::from_str(&manifest)?;
258            source_version = Some(
259                cargo["workspace"]["package"]["version"]
260                    .as_str()
261                    .context("missing workspace version")?
262                    .to_owned(),
263            );
264            continue;
265        }
266        if !include_source(product, path) {
267            continue;
268        }
269        let mut data = Vec::new();
270        item.read_to_end(&mut data)?;
271        let text = std::str::from_utf8(&data).with_context(|| format!("{path} is not UTF-8"))?;
272        validate_source(product, path, text)?;
273        let title = text
274            .lines()
275            .find_map(|line| line.strip_prefix("# "))
276            .unwrap_or_else(|| {
277                Path::new(path)
278                    .file_name()
279                    .and_then(|s| s.to_str())
280                    .unwrap_or(path)
281            })
282            .trim();
283        let corpus_path = path
284            .strip_prefix("docs/astro/src/content/docs/")
285            .map_or_else(|| path.to_owned(), |relative| format!("guides/{relative}"));
286        let id = format!("{product}/{corpus_path}");
287        let hash =
288            Sha256::digest(&data)
289                .iter()
290                .fold(String::with_capacity(64), |mut output, byte| {
291                    write!(output, "{byte:02x}").expect("writing to a string cannot fail");
292                    output
293                });
294        inventory.documents.push(json!({
295            "id": id,
296            "product": product,
297            "path": path,
298            "title": title,
299            "sha256": hash
300        }));
301        inventory.contents.insert(id, data);
302    }
303    ensure!(
304        inventory.documents.len() > before,
305        "{product}: no documents found"
306    );
307    let version = source_version.context("missing release workspace manifest")?;
308    if let Some(tag) = &source.tag {
309        ensure!(
310            *tag == format!("v{version}"),
311            "{product}: release tag {tag} does not match workspace version {version}"
312        );
313    }
314    inventory.products.insert(
315        product.to_owned(),
316        json!({
317            "version": version,
318            "revision": source.revision,
319            "status": if source.tag.is_some() { "tagged source" } else { "development snapshot" },
320            "tag": source.tag,
321            "repository": format!("https://github.com/Interchouette-ITC/{repository}")
322        }),
323    );
324    Ok(())
325}

327fn main() -> anyhow::Result<()> {
328    let root = Path::new(env!("CARGO_MANIFEST_DIR"))
329        .ancestors()
330        .nth(2)
331        .context("refresh-docs must be under tools/")?;
332    let serenade = Source {
333        tag: None,
334        revision: resolve_commit("Serenade", "dev")?,
335    };
336    let rangular = Source {
337        tag: None,
338        revision: resolve_commit("rangular", "dev")?,
339    };
340    let parser_release = latest_release("rangular")?;
341    let version = parser_release
342        .tag
343        .as_deref()
344        .context("missing Rangular release tag")?
345        .strip_prefix('v')
346        .context("invalid Rangular tag")?;
347    let mut inventory = Inventory::default();
348    refresh_product("serenade", "Serenade", &serenade, &mut inventory)?;
349    refresh_product("rangular", "rangular", &rangular, &mut inventory)?;
350    update_parser(root, version)?;
351    check_published_sources(root, &parser_release.revision, version)?;
352    let Inventory {
353        products,
354        mut documents,
355        contents,
356    } = inventory;
357    documents.sort_by(|left, right| left["id"].as_str().cmp(&right["id"].as_str()));
358    let output = root.join("corpus/documents");
359    let manifest_path = root.join("corpus/manifest.json");
360    if manifest_path.exists() {
361        let previous: Value = serde_json::from_slice(&fs::read(&manifest_path)?)?;
362        for item in previous["documents"]
363            .as_array()
364            .context("missing previous documents")?
365        {
366            let id = item["id"]
367                .as_str()
368                .context("missing previous document ID")?;
369            ensure!(safe_path(id), "unsafe old document path: {id}");
370            if !contents.contains_key(id) {
371                fs::remove_file(output.join(id))
372                    .with_context(|| format!("remove obsolete {id}"))?;
373            }
374        }
375    }
376    for (id, data) in &contents {
377        let path = output.join(id);
378        fs::create_dir_all(path.parent().context("document parent")?)?;
379        fs::write(&path, data).with_context(|| format!("write {}", path.display()))?;
380    }
381    let manifest = json!({"products": products, "documents": documents});
382    let mut serialized = serde_json::to_string_pretty(&manifest)?;
383    serialized.push('\n');
384    fs::write(manifest_path, serialized)?;
385    println!(
386        "Refreshed {} documents from Serenade dev and Rangular dev; parser uses Rangular {}.",
387        contents.len(),
388        parser_release
389            .tag
390            .as_deref()
391            .context("missing Rangular release tag")?
392    );
393    Ok(())
394}
