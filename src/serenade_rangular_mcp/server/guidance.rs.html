154pub fn guide(task: CodingTask, profile: Profile) -> Result<Value, ErrorData> {
155    let (products, summary, steps, sources) = details(task);
156    for product in products {
157        if !profile.includes(product) {
158            return Err(ErrorData::invalid_params(
159                format!("task requires the {product} profile"),
160                None,
161            ));
162        }
163    }
164    let documents: Vec<_> = sources
165        .iter()
166        .map(|id| {
167            corpus::find(id).map(corpus::entry).ok_or_else(|| {
168                ErrorData::internal_error(format!("coding guide source missing: {id}"), None)
169            })
170        })
171        .collect::<Result<_, _>>()?;
172    Ok(json!({
173        "summary": summary,
174        "steps": steps,
175        "sources": documents,
176        "scope": "Guidance only; run commands yourself or use the local stdio project tools. Development docs may describe unreleased changes."
177    }))
178}
