6fn dependencies(
7    value: &toml::Value,
8    scope: &str,
9    profile: Profile,
10    found: &mut Vec<Value>,
11) -> Result<(), ErrorData> {
12    let table = value
13        .as_table()
14        .ok_or_else(|| ErrorData::invalid_params(format!("{scope} must be a TOML table"), None))?;
15    for (alias, declaration) in table {
16        let package = match declaration {
17            toml::Value::String(_) => alias.as_str(),
18            toml::Value::Table(fields) => match fields.get("package") {
19                Some(value) => value.as_str().ok_or_else(|| {
20                    ErrorData::invalid_params(
21                        format!("{scope}.{alias}.package must be a string"),
22                        None,
23                    )
24                })?,
25                None => alias.as_str(),
26            },
27            _ => {
28                return Err(ErrorData::invalid_params(
29                    format!("{scope}.{alias} must be a version string or table"),
30                    None,
31                ));
32            }
33        };
34        let product = if package == "rangular" || package.starts_with("rangular-") {
35            "rangular"
36        } else if package.starts_with("serenade-") {
37            "serenade"
38        } else {
39            continue;
40        };
41        if profile.includes(product) {
42            found.push(json!({
43                "product": product,
44                "package": package,
45                "alias": alias,
46                "scope": scope,
47                "declaration": declaration,
48            }));
49        }
50    }
51    Ok(())
52}

54pub fn inspect(source: &str, profile: Profile) -> Result<Value, ErrorData> {
55    if source.len() > 64_000 {
56        return Err(ErrorData::invalid_params(
57            "manifest exceeds 64,000 bytes",
58            None,
59        ));
60    }
61    let manifest: toml::Value = toml::from_str(source)
62        .map_err(|error| ErrorData::invalid_params(format!("invalid TOML: {error}"), None))?;
63    let mut found = Vec::new();
64    for key in ["dependencies", "dev-dependencies", "build-dependencies"] {
65        if let Some(value) = manifest.get(key) {
66            dependencies(value, key, profile, &mut found)?;
67        }
68    }
69    if let Some(value) = manifest
70        .get("workspace")
71        .and_then(|v| v.get("dependencies"))
72    {
73        dependencies(value, "workspace.dependencies", profile, &mut found)?;
74    }
75    if let Some(targets) = manifest.get("target") {
76        let targets = targets
77            .as_table()
78            .ok_or_else(|| ErrorData::invalid_params("target must be a TOML table", None))?;
79        for (target, sections) in targets {
80            if !sections.is_table() {
81                return Err(ErrorData::invalid_params(
82                    format!("target.{target} must be a TOML table"),
83                    None,
84                ));
85            }
86            for key in ["dependencies", "dev-dependencies", "build-dependencies"] {
87                if let Some(value) = sections.get(key) {
88                    dependencies(
89                        value,
90                        &format!("target.{target}.{key}"),
91                        profile,
92                        &mut found,
93                    )?;
94                }
95            }
96        }
97    }
98    if found.len() > 100 {
99        return Err(ErrorData::invalid_params(
100            "manifest contains more than 100 matching dependencies",
101            None,
102        ));
103    }
104    Ok(json!({
105        "package": manifest.get("package").and_then(|v| v.get("name")).and_then(toml::Value::as_str),
106        "dependencies": found,
107        "scope": "Declarations in the supplied manifest only; workspace-inherited dependencies and Cargo's resolved graph require their own manifests or cargo metadata."
108    }))
109}
